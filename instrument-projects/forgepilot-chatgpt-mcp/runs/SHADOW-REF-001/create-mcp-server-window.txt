function createMcpServer(): McpServer {
  const mcpServer = new McpServer({
    name: "forgepilot-chatgpt-mcp",
    version: "0.1.0"
  });

  mcpServer.registerTool(
    "forgepilot_status",
    {
      description: "Read current ForgePilot repository status. This tool is read-only.",
      inputSchema: {},
      outputSchema: ForgePilotStatusOutputSchema,
      annotations: {
        readOnlyHint: true,
        destructiveHint: false,
        idempotentHint: true,
        openWorldHint: false
      }
    },
    async () =>
      runLoggedTool("forgepilot_status", async () => {
        const branch = await git(["branch", "--show-current"]);
        const commit = await git(["rev-parse", "--short", "HEAD"]);
        const gitStatusShort = await git(["status", "--short"]);

        const latestPackets = (await listDir("packets"))
          .filter((file) => file.endsWith(".md"))
          .sort()
          .slice(-10);

        const latestRuns = (await listDir("runs"))
          .sort()
          .slice(-10);

        const result = {
          repo: "ForgePilot",
          repoPath: REPO,
          branch,
          commit,
          workingTreeClean: gitStatusShort.length === 0,
          gitStatusShort,
          latestPackets,
          latestRuns
        };

        return structuredJsonResult(result);
      })
  );

        idempotentHint: true,
        openWorldHint: false
      }

        const text = await readRepoTextFile(safeRelativePath);
        const limit = clampMaxChars(maxChars);
        const truncated = text.length > limit;
        const content = truncated ? text.slice(0, limit) : text;

        const result = {
          path: safeRelativePath,
          content,
          truncated,
          returnedChars: content.length,
          totalChars: text.length
        };

        return structuredJsonResult(result);
      })
  );

function getHeader(req: IncomingMessage, name: string): string | undefined {
  const value = req.headers[name.toLowerCase()];
  if (Array.isArray(value)) return value[0];
  return value;
}
