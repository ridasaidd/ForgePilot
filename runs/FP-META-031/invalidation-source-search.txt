function validateEnum(value: string, validValues: string[], fieldName: string): void {
  if (!validValues.includes(value)) {
    throw new Error(
      `Invalid ${fieldName}: '${value}'. Must be one of: ${validValues.join(", ")}`
    );
  }
}

function validateObservationExists(
  targetType: string,
  targetId: number
): void {
  const db = getDb();
  let row: unknown;
  if (targetType === "classification_observation") {
    row = db
      .prepare(
        "SELECT classification_id FROM packet_classification_observations WHERE classification_id = ?"
      )
      .get(targetId);
    if (!row) {
      throw new Error(
        `Target classification_observation ${targetId} not found`
      );
    }
  } else if (targetType === "outcome_observation") {
    row = db
      .prepare(
        "SELECT outcome_id FROM model_outcome_observations WHERE outcome_id = ?"
      )
      .get(targetId);
    if (!row) {
      throw new Error(
        `Target outcome_observation ${targetId} not found`
      );
    }
  }
}

export function recordEvidenceAdmission(
  params: RecordAdmissionParams
): EvidenceAdmissionEvent {
  validateEnum(
    params.target_observation_type,
    VALID_OBSERVATION_TYPES,
    "target_observation_type"
  );
  validateEnum(
    params.admission_decision,
    VALID_ADMISSION_DECISIONS,
    "admission_decision"
  );
  validateEnum(
    params.admission_actor_type,
    VALID_ACTOR_TYPES,
    "admission_actor_type"
  );

  if (params.admission_trust_tier) {
    validateEnum(
      params.admission_trust_tier,
      VALID_TRUST_TIERS,
      "admission_trust_tier"
    );
  }
  if (params.validation_state) {
    validateEnum(
      params.validation_state,
      VALID_VALIDATION_STATES,
      "validation_state"
    );
  }

  validateObservationExists(
    params.target_observation_type,
    params.target_observation_id
  );

  const db = getDb();
  const stmt = db.prepare(
    `INSERT INTO evidence_admission_events (
      target_observation_type, target_observation_id, admission_decision,
      admission_actor_type, admission_actor_id, admission_trust_tier,
      validation_state, provenance_complete, admission_basis
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`
  );

  const result = stmt.run(
    params.target_observation_type,
    params.target_observation_id,
    params.admission_decision,
    params.admission_actor_type,
    params.admission_actor_id ?? "",
    params.admission_trust_tier ?? "TIER_0",
    params.validation_state ?? "VALID",
    params.provenance_complete ? 1 : 0,
    params.admission_basis ?? ""
  );

  return db
    .prepare("SELECT * FROM evidence_admission_events WHERE id = ?")
    .get(result.lastInsertRowid) as EvidenceAdmissionEvent;
}

  const admission = db
    .prepare("SELECT id FROM evidence_admission_events WHERE id = ?")
    .get(params.target_admission_event_id) as { id: number } | undefined;

  if (!admission) {
    throw new Error(
      `Target admission event ${params.target_admission_event_id} not found`
    );
  }

  const reviewRequest = db
    .prepare("SELECT id, target_admission_event_id FROM admission_review_requests WHERE id = ?")
    .get(params.review_request_id) as
      | { id: number; target_admission_event_id: number }
      | undefined;

  if (!reviewRequest) {
    throw new Error(
      `Review request ${params.review_request_id} not found`
    );
  }

  if (reviewRequest.target_admission_event_id !== params.target_admission_event_id) {
    throw new Error(
      `Review request ${params.review_request_id} targets admission event ` +
      `${reviewRequest.target_admission_event_id}, not ${params.target_admission_event_id}`
    );
  }

  const stmt = db.prepare(
    `INSERT INTO admission_invalidation_events (
      target_admission_event_id, review_request_id, invalidation_decision,
      invalidated_by_actor_type, invalidated_by_actor_id,
      invalidation_reason, validation_state, provenance_complete
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)`
  );

  const result = stmt.run(
    params.target_admission_event_id,
    params.review_request_id,
    params.invalidation_decision,
    params.invalidated_by_actor_type,
    params.invalidated_by_actor_id ?? "",
    params.invalidation_reason ?? "",
    params.validation_state ?? "VALID",
    params.provenance_complete ? 1 : 0
  );

  return db
    .prepare("SELECT * FROM admission_invalidation_events WHERE id = ?")
    .get(result.lastInsertRowid) as AdmissionInvalidationEvent;
}

function isAdmissionEffective(event: EvidenceAdmissionEvent): boolean {
  return (
    event.admission_decision === "ADMITTED" &&
    event.validation_state === "VALID" &&
    event.provenance_complete === 1
  );
}

function isInvalidationDefeating(event: AdmissionInvalidationEvent): boolean {
  return (
    event.validation_state === "VALID" &&
    event.provenance_complete === 1 &&
    (event.invalidation_decision === "INVALIDATED" ||
      event.invalidation_decision === "QUARANTINED")
  );
}

export function deriveEvidenceEligibility(
  targetObservationType: string,
  targetObservationId: number
): EvidenceEligibilityResult {
  const db = getDb();

  const admissions = db
    .prepare(
      `SELECT * FROM evidence_admission_events
       WHERE target_observation_type = ? AND target_observation_id = ?
       ORDER BY created_at ASC, id ASC`
    )
    .all(targetObservationType, targetObservationId) as EvidenceAdmissionEvent[];

  let lastEffectiveAdmission: EvidenceAdmissionEvent | null = null;
  let lastDefeatingInvalidation: AdmissionInvalidationEvent | null = null;

  for (const admission of admissions) {
    if (!isAdmissionEffective(admission)) {
      continue;
    }

    lastEffectiveAdmission = admission;
    lastDefeatingInvalidation = null;

    const invalidations = db
      .prepare(
        `SELECT * FROM admission_invalidation_events
         WHERE target_admission_event_id = ?
         ORDER BY created_at ASC, id ASC`
      )
      .all(admission.id) as AdmissionInvalidationEvent[];

    let defeated = false;
    for (const invalidation of invalidations) {
      if (isInvalidationDefeating(invalidation)) {
        defeated = true;
        lastDefeatingInvalidation = invalidation;
        break;
      }
    }

    if (!defeated) {
      return {
        eligible: true,
        admission_event: admission,
        defeating_invalidation: null,
      };
    }
  }

  return {
    eligible: false,
    admission_event: lastEffectiveAdmission,
    defeating_invalidation: lastDefeatingInvalidation,
  };
}


## Admission reconstruction/report excerpts
import fs from "node:fs";
import path from "node:path";

export const RECONSTRUCTION_BOUNDARY =
  "Reconstruction is read-only. It summarizes recorded state and does not create admission, comparison, ranking, or routing authority.";

export interface ReconstructionArtifact {
  path: string;
  kind: "verification" | "audit" | "comparison" | "admission" | "executor_result";
}

export interface ReconstructionReport {
  packet_id: string;
  packet_path: string;
  packet_exists: boolean;
  source_artifacts: ReconstructionArtifact[];
  verification_state: "PRESENT" | "MISSING";
  audit_state: "PRESENT" | "MISSING";
  comparison_state: "PRESENT" | "NO_COMPARISON_RECORDED";
  admission_state: "PRESENT" | "NOT_EVALUATED" | "UNKNOWN";
  reconstructed_labels: string[];
  boundary: string;
  caveats: string[];
}

interface BuildReconstructionReportOptions {
  packetId: string;
  rootDir?: string;
}

function cleanPacketId(packetId: string): string {
  const trimmed = packetId.trim();
  if (!trimmed) throw new Error("packetId is required");
  if (trimmed.includes("/") || trimmed.includes("\\") || trimmed.includes("..")) {
    throw new Error(`invalid packetId: ${packetId}`);
  }
  return trimmed;
}

function walkFiles(dir: string): string[] {
  if (!fs.existsSync(dir)) return [];
  const files: string[] = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) files.push(...walkFiles(full));
    if (entry.isFile()) files.push(full);
  }
  return files;
}

function kindFor(rel: string): ReconstructionArtifact["kind"] | null {
  const base = path.basename(rel).toLowerCase();
  if (base.includes("verification")) return "verification";
  if (base.includes("audit")) return "audit";
  if (base.includes("comparison")) return "comparison";
  if (base.includes("admission")) return "admission";
  if (base === "executor-result.md") return "executor_result";
  return null;
}

export function buildReconstructionReport(options: BuildReconstructionReportOptions): ReconstructionReport {
  const packetId = cleanPacketId(options.packetId);
  const rootDir = options.rootDir ?? process.cwd();
  const packetPath = `packets/${packetId}.md`;
  const absolutePacketPath = path.join(rootDir, packetPath);
  if (!fs.existsSync(absolutePacketPath)) {
    throw new Error(`packet not found: ${packetPath}`);
  }

  const runDir = path.join(rootDir, "runs", packetId);
  const source_artifacts = walkFiles(runDir)
    .map((full) => path.relative(rootDir, full).split(path.sep).join("/"))
    .filter((rel) => !rel.includes("/terminal-commands/"))
    .map((rel) => ({ path: rel, kind: kindFor(rel) }))
    .filter((entry): entry is ReconstructionArtifact => entry.kind !== null)
    .sort((a, b) => a.path.localeCompare(b.path));

  const has = (kind: ReconstructionArtifact["kind"]) => source_artifacts.some((a) => a.kind === kind);
  const verification_state = has("verification") ? "PRESENT" : "MISSING";
  const audit_state = has("audit") ? "PRESENT" : "MISSING";
  const comparison_state = has("comparison") ? "PRESENT" : "NO_COMPARISON_RECORDED";
  const admission_state = has("admission") ? "PRESENT" : "NOT_EVALUATED";

  const reconstructed_labels = [
    verification_state === "PRESENT" ? "VERIFICATION_ARTIFACT_PRESENT" : "VERIFICATION_MISSING",
    audit_state === "PRESENT" ? "AUDIT_ARTIFACT_PRESENT" : "AUDIT_MISSING",
    comparison_state === "PRESENT" ? "COMPARISON_ARTIFACT_PRESENT" : "NO_COMPARISON_RECORDED",
    admission_state === "PRESENT" ? "ADMISSION_ARTIFACT_PRESENT" : "NOT_EVALUATED",
  ];

  return {
    packet_id: packetId,
    packet_path: packetPath,
    packet_exists: true,
    source_artifacts,
    verification_state,
    audit_state,
    comparison_state,
    admission_state,
    reconstructed_labels,
    boundary: RECONSTRUCTION_BOUNDARY,
    caveats: [
      RECONSTRUCTION_BOUNDARY,
      "Artifact presence does not imply admission or correctness.",
      "Comparison artifacts are bounded records, not global model rankings.",
      "Missing comparison is reported as missing, not as failure.",
    ],
  };
}

export function formatReconstructionReportText(report: ReconstructionReport): string {
  const lines = [
    `Reconstruction Report: ${report.packet_id}`,
    `Packet Path: ${report.packet_path}`,
    `Verification State: ${report.verification_state}`,
    `Audit State: ${report.audit_state}`,
    `Comparison State: ${report.comparison_state}`,
    `Admission State: ${report.admission_state}`,
    `Labels: ${report.reconstructed_labels.join(", ")}`,
    `Boundary: ${report.boundary}`,
    "Caveats:",
    ...report.caveats.map((c) => `  - ${c}`),
  ];
  if (report.source_artifacts.length > 0) {
    lines.push("Source Artifacts:");
    lines.push(...report.source_artifacts.map((a) => `  - ${a.path} [${a.kind}]`));
  }
  return `${lines.join("\n")}\n`;
}
# Admission and Comparison Reconstruction Rules

## Purpose

ForgePilot reports need to reconstruct admission and comparison state without creating trust by summary.

Reconstruction is read-only.

It summarizes recorded state; it does not create admission, invalidate evidence, rank models globally, or route future work.

## Governing Question

```text
How may ForgePilot reconstruct admission and comparison state for reports?
```

## Reconstruction Inputs

A reconstruction report may use only committed artifacts or supported read-only query results.

Allowed inputs:

| Input | Use |
|---|---|
| `packets/<packet>.md` | packet intent and acceptance criteria |
| `runs/<packet>/verification*` | recorded verification output |
| `runs/<packet>/audit*` | recorded audit output when present |
| `runs/<packet>/comparison*` | recorded comparison result when present |
| `runs/<packet>/executor-result.md` | executor result artifact when present |
| `runs/<packet>/metrics.json` | observation only unless admitted |
| `runs/<packet>/**/telemetry.json` | observation only unless admitted |
| SQLite read-only query results | recorded persisted state only |
| `docs/forgepilot-progress-checklist.md` | checklist-derived index status only |

## Admission State Reporting Rules

Reports may show admission state exactly as recorded.

Allowed admission labels:

- `ADMITTED`
- `REJECTED`
- `PENDING`
- `QUARANTINED`
- `NOT_EVALUATED`
- `UNKNOWN`

If admission state is missing, the report must use `UNKNOWN` or `NOT_EVALUATED`, depending on the source.

A report must not infer `ADMITTED` from:

- passing tests,
- accepted audit text,
- committed metrics,
- telemetry existence,
- comparison result existence,
- checklist status.

## Comparison State Reporting Rules

Reports may summarize comparison artifacts only as bounded comparison records.

Allowed comparison labels:

- `SELECTED`
- `NOT_SELECTED`
- `TIE`
- `ACCEPTED`
- `REJECTED`
- `NO_COMPARISON_RECORDED`
- `UNKNOWN`

A comparison report must identify the comparison source artifact or query.

A comparison must not be treated as global model ranking.

## Source Precedence

When sources disagree, use this order:

1. Explicit admission/comparison database state from read-only query.
2. Committed admission/comparison artifact under `runs/<packet>/`.
3. Packet-specific verification/audit artifacts.
4. Progress checklist entry.
5. Packet existence.

Lower-precedence sources may provide context, but must not override higher-precedence state.

## Unknown and Missing State Handling

Missing evidence is not failure by itself.

Reports must distinguish:

| State | Meaning |
|---|---|
| `UNKNOWN` | source could not determine state |
| `NOT_EVALUATED` | source explicitly says no evaluation/admission has occurred |
| `NO_COMPARISON_RECORDED` | no comparison artifact or state was found |
| `PARTIAL` | some evidence exists but reconstruction is incomplete |
| `DEFERRED` | intentionally postponed |

## Historical Packet Handling

Historical packets remain valid historical evidence.

Reports may say:

```text
This packet predates current taxonomy or reconstruction rules.
```

Reports must not rename historical packet IDs or reinterpret old packet names as errors.

## Prohibited Inferences

Reconstruction reports must not claim:

- verification equals admission,
- audit acceptance equals admission,
- telemetry existence equals evidence value,
- comparison equals global model ranking,
- selected implementation means future routing preference,
- missing comparison means failure,
- historical namespace mismatch means packet invalidity,
- checklist status is proof by itself.

## Implementation Constraints for Later `FP-*` Packet

A later implementation should be read-only by default.

It should:

- collect admission/comparison candidates from artifacts and read-only queries,
- preserve source paths,
- report unknown state explicitly,
- include source precedence in output,
- include prohibited-inference caveats,
- avoid mutation or admission changes.

It must not:

- write admission events,
- update comparison outcomes,
- rewrite historical artifacts,
- rank models globally,
- route future work.

## Required Boundary Text

Any report using reconstructed admission/comparison state must include:

```text
Reconstruction is read-only. It summarizes recorded state and does not create admission, comparison, ranking, or routing authority.
```
